6. PROBAR LOS IDS DEL FOLLOWER:

python - <<'PY'
import time
from lerobot.robots.so_follower import SO101Follower, SO101FollowerConfig

robot = SO101Follower(SO101FollowerConfig(port="/dev/ttyACM0", id="follower_2"))
bus = robot.bus
bus.connect()
try:
    print("Asignación configurada:")
    for nombre, motor in bus.motors.items():
        print(f"  ID {motor.id}: {nombre}")

    activos = [
        nombre for nombre in bus.motors
        if bus.read("Torque_Enable", nombre, normalize=False) != 0
    ]
    if activos:
        raise SystemExit(f"Torque activo en {activos}. Detén la prueba sin moverlos a mano.")

    print("\nDeja el brazo quieto. Tomando posición de referencia...")
    time.sleep(2)
    inicio = bus.sync_read("Present_Position", normalize=False)
    print("Mueve una articulación por vez y devuélvela al inicio. Ctrl+C para salir.")

    while True:
        actual = bus.sync_read("Present_Position", normalize=False)
        cambios = []
        for nombre, valor in actual.items():
            delta = valor - inicio[nombre]
            if abs(delta) >= 25:
                cambios.append(f"ID {bus.motors[nombre].id} {nombre}: {delta:+d}")
        print(" | ".join(cambios) if cambios else "En posición de referencia", flush=True)
        time.sleep(0.4)
except KeyboardInterrupt:
    pass
finally:
    bus.disconnect(False)
PY
